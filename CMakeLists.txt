cmake_minimum_required(VERSION 3.16)
project(forge_postgres VERSION 0.1.2 LANGUAGES C)
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
add_library(forge_postgres STATIC src/bridge.c)
target_include_directories(forge_postgres PUBLIC include ${PostgreSQL_INCLUDE_DIRS})
target_link_libraries(forge_postgres PUBLIC ${PostgreSQL_LIBRARIES} Threads::Threads)
target_compile_features(forge_postgres PUBLIC c_std_11)
target_compile_options(forge_postgres PRIVATE -Wall -Wextra -Werror)

include(CTest)
if(BUILD_TESTING)
  add_executable(prepared_test tests/prepared.c)
  target_link_libraries(prepared_test PRIVATE forge_postgres)
  target_compile_options(prepared_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME prepared_query_integration COMMAND prepared_test)
  set_tests_properties(prepared_query_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 45)
endif()
