cmake_minimum_required(VERSION 3.16)
project(forge_postgres VERSION 0.1.0 LANGUAGES C)
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
add_library(forge_postgres STATIC src/bridge.c)
target_include_directories(forge_postgres PUBLIC include ${PostgreSQL_INCLUDE_DIRS})
target_link_libraries(forge_postgres PUBLIC ${PostgreSQL_LIBRARIES} Threads::Threads)
target_compile_features(forge_postgres PUBLIC c_std_11)
target_compile_options(forge_postgres PRIVATE -Wall -Wextra -Werror)
